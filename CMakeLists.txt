cmake_minimum_required(VERSION 3.15)

get_property(_is_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)

if(_is_multi_config)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release;ReleaseWithProfiling"
        CACHE STRING "Available configurations" FORCE)
endif()

project(SoftXSolution LANGUAGES CXX)

if(_is_multi_config)
    set(CMAKE_CXX_FLAGS_RELEASEWITHPROFILING
        "${CMAKE_CXX_FLAGS_RELEASE}" CACHE STRING "" FORCE)
    set(CMAKE_C_FLAGS_RELEASEWITHPROFILING
        "${CMAKE_C_FLAGS_RELEASE}" CACHE STRING "" FORCE)
    set(CMAKE_EXE_LINKER_FLAGS_RELEASEWITHPROFILING
        "${CMAKE_EXE_LINKER_FLAGS_RELEASE}" CACHE STRING "" FORCE)
    set(CMAKE_SHARED_LINKER_FLAGS_RELEASEWITHPROFILING
        "${CMAKE_SHARED_LINKER_FLAGS_RELEASE}" CACHE STRING "" FORCE)
    set(CMAKE_STATIC_LINKER_FLAGS_RELEASEWITHPROFILING
        "${CMAKE_STATIC_LINKER_FLAGS_RELEASE}" CACHE STRING "" FORCE)
endif()

if(NOT _is_multi_config AND CMAKE_BUILD_TYPE STREQUAL "ReleaseWithProfiling")
    set(CMAKE_CXX_FLAGS_RELEASEWITHPROFILING "${CMAKE_CXX_FLAGS_RELEASE}")
    set(CMAKE_EXE_LINKER_FLAGS_RELEASEWITHPROFILING "${CMAKE_EXE_LINKER_FLAGS_RELEASE}")
endif()

add_subdirectory(Source/Third-Party/Optick)
add_subdirectory(Source/SoftX)
add_subdirectory(Samples)
